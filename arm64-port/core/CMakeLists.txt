cmake_minimum_required(VERSION 3.16)
project(PicSayImageEngine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CORE_SOURCES src/ImageEngine.cpp src/Effects.cpp)
add_library(PicSayImageEngine STATIC ${CORE_SOURCES})
target_include_directories(PicSayImageEngine PUBLIC include)
if(ANDROID)
 add_library(picsay_engine SHARED src/AndroidBitmapBridge.cpp ${CORE_SOURCES})
 target_include_directories(picsay_engine PRIVATE include)
 find_library(android-lib android)
 find_library(jnigraphics-lib jnigraphics)
 target_link_libraries(picsay_engine PRIVATE ${android-lib} ${jnigraphics-lib})
endif()
option(PICSAY_BUILD_TESTS "Build host unit tests" ON)
if(PICSAY_BUILD_TESTS AND NOT ANDROID)
 enable_testing()
 add_executable(image_engine_tests tests/ImageEngineTests.cpp)
 target_link_libraries(image_engine_tests PRIVATE PicSayImageEngine)
 add_test(NAME image_engine_tests COMMAND image_engine_tests)
 add_executable(effects_tests tests/EffectsTests.cpp)
 target_link_libraries(effects_tests PRIVATE PicSayImageEngine)
 add_test(NAME effects_tests COMMAND effects_tests)
endif()
